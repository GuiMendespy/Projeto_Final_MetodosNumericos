\subsection{Etapa 2: Aplicação do Método}

\begin{lstlisting}[caption={Decomposição / aplicação do método}, label={lst:method}]
from scipy.linalg import svd

# Decomposicao SVD truncada em k componentes
k = 50
U, s, Vt = svd(A_norm, full_matrices=False)

U_k  = U[:, :k]
S_k  = np.diag(s[:k])
Vt_k = Vt[:k, :]

A_approx = U_k @ S_k @ Vt_k
print(f"Erro de reconstrucao (Frobenius): {np.linalg.norm(A - A_approx):.4f}")
\end{lstlisting}
